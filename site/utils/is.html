使用示例 ​

基础类型判断 ​

ts
import { 
    isUndefined, isNull, isBoolean, isNumber, 
    isString, isFunction, isArray, isObject 
} from '@wlink/utils'

// 基础类型
isUndefined(undefined)  // true
isNull(null)           // true
isBoolean(true)        // true
isNumber(42)           // true
isString('hello')      // true
isFunction(() => {})   // true
isArray([1, 2, 3])     // true
isObject({ a: 1 })     // true

// 注意：isNumber 会排除 NaN
isNumber(NaN)          // false

复杂类型判断 ​

ts
import { isDate, isRegExp, isPromise, isSymbol, isBigInt } from '@wlink/utils'

// 日期类型（必须是有效日期）
isDate(new Date())              // true
isDate(new Date('2023-01-01'))  // true
isDate(new Date('invalid'))     // false

// 正则表达式
isRegExp(/test/)               // true
isRegExp(new RegExp('test'))   // true

// Promise
isPromise(Promise.resolve())   // true
isPromise({ then: () => {} })  // true (thenable)

// Symbol 和 BigInt
isSymbol(Symbol('test'))       // true
isBigInt(123n)                 // true

深度相等判断 ​

ts
import { isEqual } from '@wlink/utils'

// 基础类型
isEqual(1, 1)                   // true
isEqual('a', 'a')               // true

// 对象比较
isEqual({ a: 1 }, { a: 1 })     // true
isEqual({ a: 1, b: 2 }, { b: 2, a: 1 }) // true

// 数组比较
isEqual([1, 2, 3], [1, 2, 3])   // true

// 嵌套结构
const obj1 = { a: { b: [1, 2] } }
const obj2 = { a: { b: [1, 2] } }
isEqual(obj1, obj2)             // true

// 不相等的情况
isEqual({ a: 1 }, { a: 2 })     // false
isEqual([1, 2], [1, 2, 3])      // false

实际应用场景 ​

表单验证 ​

ts
import { isString, isNumber, isEmpty, isBlank } from '@wlink/utils'

function validateForm(data: any) {
    const errors: string[] = []
    
    if (!isString(data.name) || isBlank(data.name)) {
        errors.push('姓名不能为空')
    }
    
    if (!isNumber(data.age) || data.age < 0) {
        errors.push('年龄必须是有效数字')
    }
    
    if (!isEmpty(data.email) && !isString(data.email)) {
        errors.push('邮箱格式不正确')
    }
    
    return errors
}

数据处理 ​

ts
import { isArray, isObject, isEmpty } from '@wlink/utils'

function processData(data: any) {
    if (isEmpty(data)) {
        return null
    }
    
    if (isArray(data)) {
        return data.filter(item => !isEmpty(item))
    }
    
    if (isObject(data)) {
        const result: any = {}
        for (const [key, value] of Object.entries(data)) {
            if (!isEmpty(value)) {
                result[key] = value
            }
        }
        return result
    }
    
    return data
}

类型守卫 ​

ts
import { isString, isNumber, isArray } from '@wlink/utils'

function processValue(value: unknown) {
    if (isString(value)) {
        // TypeScript 知道这里 value 是 string 类型
        return value.toUpperCase()
    }
    
    if (isNumber(value)) {
        // TypeScript 知道这里 value 是 number 类型
        return value.toFixed(2)
    }
    
    if (isArray(value)) {
        // TypeScript 知道这里 value 是 array 类型
        return value.length
    }
    
    return 'unknown type'
}

注意事项 ​

  1. NaN 处理：isNumber() 会排除 NaN，使用 isValidNumber() 获得更严格的数字验证
  2. 日期验证：isDate() 不仅检查类型，还验证日期的有效性
  3. 对象判断：isObject() 只判断普通对象，不包括数组、函数、日期等
  4. 类型守卫：所有判断函数都提供了 TypeScript 类型守卫功能
  5. 性能考虑：对于大量数据的类型检查，建议缓存检查结果

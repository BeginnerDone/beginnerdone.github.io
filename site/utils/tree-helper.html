介绍 ​

常用于tree的常用操作

方法 ​

listToTree ​

集合转为树结构，通过对应的parentId进行转换

示例

ts
 const list = [
        { id: 1, parentId: null },
        { id: 2, parentId: 1 },
        { id: 3, parentId: 1 },
        { id: 4, parentId: 2 },
        { id: 5, parentId: 2 }
    ];

   listToTree(list,{
       id: "id",
       pid: "parentId",
       children: "children"
   })

  // 最后输出 tree
  [
        {
            id: 1,
            parentId: null,
            children: [
                {
                    id: 2,
                    parentId: 1,
                    children: [
                        { id: 4, parentId: 2 },
                        { id: 5, parentId: 2 }
                    ]
                },
                { id: 3, parentId: 1 }
            ]
        }
   ];

treeToList ​

将树结构转为集合，通过parentId

ts
 const tree = [
        {
            id: 1,
            parentId: null,
            children: [
                {
                    id: 2,
                    parentId: 1,
                    children: [
                        { id: 4, parentId: 2 },
                        { id: 5, parentId: 2 }
                    ]
                },
                { id: 3, parentId: 1 }
            ]
        }
   ];
   treeToList(list,{
       id: "id",
       pid: "parentId",
       children: "children"
   })

  // 最后输出 list
   [
        { id: 1, parentId: null },
        { id: 2, parentId: 1 },
        { id: 3, parentId: 1 },
        { id: 4, parentId: 2 },
        { id: 5, parentId: 2 }
    ];

findNodeAll ​

用于在树形结构中查找所有满足特定条件的节点的集合。

ts
 const tree = [
        {
            id: 1,
            parentId: null,
            children: [
                {
                    id: 2,
                    parentId: 1,
                    children: [
                        { id: 4, parentId: 2 },
                        { id: 5, parentId: 2 }
                    ]
                },
                { id: 3, parentId: 1 }
            ]
        }
   ];
findNodeAll(tree, (node: any) => node.id > 2,{
       id: "id",
       pid: "parentId",
       children: "children"
 })
// 输出
【{ id: 4, parentId: 2 },{ id: 5, parentId: 2 }, { id: 3, parentId: 1 }】

findNode ​

它在树形数据结构中查找满足特定条件的第一个节点

ts
const tree = [
    {
        id: 1,
        parentId: null,
        children: [
            {
                id: 2,
                parentId: 1,
                children: [
                    { id: 4, parentId: 2 },
                    { id: 5, parentId: 2 }
                ]
            },
            { id: 3, parentId: 1 }
        ]
    }
];

findNode(tree, (node: any) => node.id === 4, {
    id: "id",
    pid: "parentId",
    children: "children"
})
// 输出: { id: 4, parentId: 2 }

findPath ​

在树中找到满足给定条件的节点的路径

ts
const tree = [
    {
        id: 1,
        name: 'root',
        children: [
            {
                id: 2,
                name: 'child1',
                children: [
                    { id: 4, name: 'child2' },
                    { id: 5, name: 'child3' }
                ]
            },
            { id: 3, name: 'child4' }
        ]
    }
];

findPath(tree, (node: any) => node.id === 4, {
    id: "id",
    pid: "parentId",
    children: "children"
})
// 输出: [{ id: 1, name: 'root' }, { id: 2, name: 'child1' }, { id: 4, name: 'child2' }]

findPathAll ​

在树中找到所有满足给定条件的节点的路径

ts
const tree = [
    {
        id: 1,
        name: 'root',
        children: [
            {
                id: 2,
                name: 'child1',
                children: [
                    { id: 4, name: 'target' },
                    { id: 5, name: 'child3' }
                ]
            },
            { id: 3, name: 'target' }
        ]
    }
];

findPathAll(tree, (node: any) => node.name === 'target', {
    id: "id",
    pid: "parentId",
    children: "children"
})
// 输出: [
//   [{ id: 1, name: 'root' }, { id: 2, name: 'child1' }, { id: 4, name: 'target' }],
//   [{ id: 1, name: 'root' }, { id: 3, name: 'target' }]
// ]

filter ​

基于给定条件过滤数组树的元素

ts
const tree = [
    {
        id: 1,
        name: 'root',
        status: 'active',
        children: [
            {
                id: 2,
                name: 'child1',
                status: 'inactive',
                children: [
                    { id: 4, name: 'child2', status: 'active' },
                    { id: 5, name: 'child3', status: 'inactive' }
                ]
            },
            { id: 3, name: 'child4', status: 'active' }
        ]
    }
];

filter(tree, (node: any) => node.status === 'active', {
    id: "id",
    pid: "parentId",
    children: "children"
})
// 输出: [
//   {
//     id: 1,
//     name: 'root',
//     status: 'active',
//     children: [
//       {
//         id: 2,
//         name: 'child1',
//         status: 'inactive',
//         children: [
//           { id: 4, name: 'child2', status: 'active' }
//         ]
//       },
//       { id: 3, name: 'child4', status: 'active' }
//     ]
//   }
// ]

配置选项 ​

所有方法都支持通过 config 参数自定义字段名：

ts
interface TreeOption {
    id: string;      // 节点唯一标识字段名，默认 'id'
    pid: string;     // 父节点标识字段名，默认 'parentId'
    children: string; // 子节点数组字段名，默认 'children'
}

// 默认配置
const defaultOption: TreeOption = {
    id: "id",
    pid: "parentId",
    children: "children"
};

实际应用场景 ​

组织架构处理 ​

ts
import { listToTree, findNode, findPath } from '@wlink/utils'

// 扁平的部门数据
const departments = [
    { id: 1, name: '总公司', parentId: null },
    { id: 2, name: '技术部', parentId: 1 },
    { id: 3, name: '市场部', parentId: 1 },
    { id: 4, name: '前端组', parentId: 2 },
    { id: 5, name: '后端组', parentId: 2 }
];

// 转换为树形结构
const orgTree = listToTree(departments);

// 查找特定部门
const frontendTeam = findNode(orgTree, node => node.name === '前端组');

// 获取部门路径
const teamPath = findPath(orgTree, node => node.name === '前端组');
// 结果: ['总公司', '技术部', '前端组']

菜单权限过滤 ​

ts
import { filter } from '@wlink/utils'

const menuTree = [
    {
        id: 1,
        name: '系统管理',
        permission: 'admin',
        children: [
            { id: 2, name: '用户管理', permission: 'user:manage' },
            { id: 3, name: '角色管理', permission: 'role:manage' }
        ]
    },
    {
        id: 4,
        name: '业务模块',
        permission: 'business',
        children: [
            { id: 5, name: '订单管理', permission: 'order:view' }
        ]
    }
];

// 根据用户权限过滤菜单
const userPermissions = ['admin', 'user:manage', 'business', 'order:view'];
const filteredMenu = filter(menuTree, node => 
    userPermissions.includes(node.permission)
);

文件目录操作 ​

ts
import { findNodeAll, treeToList } from '@wlink/utils'

const fileTree = [
    {
        id: 1,
        name: 'src',
        type: 'folder',
        children: [
            {
                id: 2,
                name: 'components',
                type: 'folder',
                children: [
                    { id: 3, name: 'Button.vue', type: 'file' },
                    { id: 4, name: 'Input.vue', type: 'file' }
                ]
            },
            { id: 5, name: 'main.ts', type: 'file' }
        ]
    }
];

// 查找所有 Vue 文件
const vueFiles = findNodeAll(fileTree, node => 
    node.type === 'file' && node.name.endsWith('.vue')
);

// 将树形结构转为扁平列表（用于搜索）
const flatFileList = treeToList(fileTree);

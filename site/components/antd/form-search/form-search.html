介绍 ​

融合和 a-form组件以及antdv中其他的表单组件形成的搜素表头，该组件主要是为了a-table服务组成业务组件，当然单独使用也是可以的

基础 ​

基本的搜素
通过设置基本的 items 属性即可生成一个搜索表单
<template>
    <div>
        <w-form-search v-model:model="form" :items="items" @search="onSearch" />
        <div>
            结果：
            <pre>{{ form }}</pre>
        </div>
    </div>
</template>
<script lang="ts" setup>
import { ref } from "vue";
const items = ref([
    {
        type: "input",
        itemAttrs: {
            label: "姓名",
            name: "name"
        }
    },
    {
        type: "input",
        itemAttrs: {
            label: "年龄",
            name: "age"
        }
    },
    {
        type: "inputNumber",
        itemAttrs: {
            label: "分数",
            name: "score"
        }
    },
    {
        type: "select",
        autoSearch: true,
        itemAttrs: {
            label: "班主任",
            name: "teacher"
        },
        componentAttrs: {
            url: "select/list"
        }
    },
    {
        type: "datePicker",
        itemAttrs: {
            label: "日期",
            name: "date"
        },
        dateTimeAttrs: {
            range: true,
            startField: "startDate",
            endField: "endDate"
        }
    }
]);
const form = ref({
    name: "张三"
});
</script>

默认查询条件 ​

initValue 和 model
通过设置 initValue 或者 model 属性可以设置默认的查询条件，切记这里最好是使用 initValue 去绑定，因为 model 属性主要是为了动态改变一个值，这里 initValue 表示默认查询条件不会随着重置而改变，所以如果要重新回到空的状态，那么使用 model 绑定初始化是最好的办法
<template>
    <div>
        <a-divider orientation="left">设置items中的initValue属性</a-divider>
        <w-form-search :model="form" :items="items" @search="onSearch" />
        <a-divider orientation="left">设置model属性</a-divider>
        <w-form-search :model="form1" :items="items1" @search="onSearch" />
    </div>
</template>

前置后置按钮 ​

通过设置前置后置插槽可以在查询按钮的前后添加自定义的按钮
<template>
    <div>
        <w-form-search
            :items="items"
            @search="onSearch"
            :span="8"
            :action-span="12"
        >
            <template #addonBefore>
                <a-button type="primary">前置</a-button>
            </template>
            <template #addonAfter>
                <a-button type="primary">后置</a-button>
            </template>
        </w-form-search>
    </div>
</template>

    {
        type: "input",
        itemAttrs: {
            label: "姓名",
            name: "name"
        }
    },
    {
        type: "input",
        itemAttrs: {
            label: "年龄",
            name: "age"
        }
    },
    {
        type: "inputNumber",
        itemAttrs: {
            label: "分数",
            name: "score"
        }
    },
    {
        type: "select",
        autoSearch: true,
        itemAttrs: {
            label: "班主任",

        },
        componentAttrs: {
            url: "select/list"
        }

    {
        type: "datePicker",

动态按钮区域宽度 ​

动态按钮区域宽度
开启 `autoActionSpan` 后，组件会根据操作按钮的实际宽度自动分配 span；可通过增减扩展按钮观察操作区和首行查询条件的变化
<template>
    <div>
        <a-space style="margin-bottom: 16px">
            <a-button
                :disabled="actionCount >= MAX_ACTION_COUNT"
                @click="addAction"
            >
                增加扩展按钮
            </a-button>
            <a-button :disabled="actionCount === 0" @click="removeAction">
                减少扩展按钮
            </a-button>
            <span>当前扩展按钮：{{ actionCount }} 个</span>
        </a-space>

        <w-form-search :items="items" :span="6" auto-action-span>
            <template #addonAfter>
                <a-button v-for="index in actionCount" :key="index">
                    扩展操作 {{ index }}
                </a-button>
            </template>
        </w-form-search>
    </div>
</template>

<script lang="ts" setup>
import { ref } from "vue";
import type { ExtFormItem } from "wlink-ui/antd";

const MAX_ACTION_COUNT = 3;
const actionCount = ref(0);

const items: ExtFormItem[] = [
    {
        type: "input",
        itemAttrs: {
            label: "姓名",
            name: "name"
        }
    },
    {
        type: "input",
        itemAttrs: {
            label: "年龄",
            name: "age"
        }
    },
    {
        type: "input",
        itemAttrs: {
            label: "部门",
            name: "department"
        }
    },
    {
        type: "input",
        itemAttrs: {
            label: "状态",
            name: "status"
        }
    }
];

const addAction = () => {
    actionCount.value = Math.min(MAX_ACTION_COUNT, actionCount.value + 1);
};

const removeAction = () => {
    actionCount.value = Math.max(0, actionCount.value - 1);
};
</script>
